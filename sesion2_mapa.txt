>> 5. Antes de </body>: script con el mapa de Leaflet y OpenStreetMap

    var map = L.map('map').setView([23, -102], 5);

    L.tileLayer('https://tile.openstreetmap.org/{z}/{x}/{y}.png', {
        maxZoom: 19,
        attribution: '&copy; OpenStreetMap'
    }).addTo(map);
</script>


>> 6. Dentro del <script>: reemplaza el L.tileLayer de OpenStreetMap por el mapa base sugerido

L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Street_Map/MapServer/tile/{z}/{y}/{x}', {
    maxZoom: 17,
    attribution: 'Tiles &copy; Esri'
}).addTo(map);


>> 7. Dentro del <script>, debajo del L.tileLayer: datos GeoJSON con círculos y popups

fetch('pueblos_magicos.geojson')
    .then(respuesta => respuesta.json())
    .then(datos => {
        L.geoJSON(datos, {
            pointToLayer: (punto, coordenadas) => L.circleMarker(coordenadas, {
                radius: 7, color: 'white', weight: 2, fillColor: '#e4007c', fillOpacity: 0.9
            }),
            onEachFeature: (punto, capa) => {
                capa.bindPopup('<strong>' + punto.properties.nombre + '</strong><br>' + punto.properties.estado + '<br>' + punto.properties.atractivo);
            }
        }).addTo(map);
    });

>> 9. Dentro del <script>, al final (antes de </script>): mapa de MapLibre con mapa de calor

const mlmap = new maplibregl.Map({
    container: 'mlmap',
    style: {
        version: 8,
        sources: {
            esri: {
                type: 'raster',
                tiles: ['https://server.arcgisonline.com/ArcGIS/rest/services/Canvas/World_Dark_Gray_Base/MapServer/tile/{z}/{y}/{x}'],
                tileSize: 256,
                maxzoom: 16,
                attribution: 'Tiles &copy; Esri'
            }
        },
        layers: [{ id: 'esri', type: 'raster', source: 'esri' }]
    },
    center: [-100, 22],
    zoom: 4.3
});
mlmap.addControl(new maplibregl.NavigationControl(), 'top-right');

mlmap.on('load', () => {
    mlmap.addSource('pueblos', { type: 'geojson', data: 'pueblos_magicos.geojson' });

    mlmap.addLayer({
        id: 'mapa-de-calor',
        type: 'heatmap',
        source: 'pueblos',
        paint: {
            'heatmap-radius': 40,
            'heatmap-intensity': 1.2,
            'heatmap-opacity': 0.85,
            'heatmap-color': [
                'interpolate', ['linear'], ['heatmap-density'],
                0,   'rgba(0, 0, 0, 0)',
                0.2, '#fde0ef',
                0.5, '#f768a1',
                0.8, '#c51b8a',
                1,   '#7a0177'
            ]
        }
    });
});
